\begin{definition}
    For topology space $X$ with given \[f: X \doublearrow{} Y \] onto. $O \subset Y$ is open iff $f\inv(O) \subset X$ is open. It is called "quotient topology" on $Y$.
\end{definition}

For instance, let $X = \R$ and give equivalent relation with following $x \sim y$ iff $x-y \in \Z$.

\[
\R/{\sim} = \set{[x]_{\sim} : x \in \R}
\]

\begin{proposition}
Then, with natural map and quotient topology, $\R/{\sim} \cong S^1$.
\end{proposition}

\begin{proof}
\[
\begin{tikzcd}
\R \arrow[r, "f"] \arrow[d, "q"'] & S^1 \\
\R/{\sim} \arrow[ur, "\bar f"']
\end{tikzcd}
\]

\[
    \bar f : [x] \mapsto \exp(2\pi i x)
\]

Since, $[0, 1]$ is compact then $\R/{\sim}$ is compact. ($\because$ $q$ is conti) For open set $U \subset S^1$, $\bar f\inv(U)$ is open iff $q\inv(\bar f\inv(U))=f\inv(U)$ is open then this is continuous. $\bar f$ is clearly bijection, then by \cref{proposition:homeo}, it is homeomorphism.

\end{proof}

\begin{proposition}
    Let $X, Y$ is Hausdorff compact topology space. If there exists 1-1 correspondence conti map \[f: X \to Y\], then homeomorphic.
    \label{proposition:homeo}
\end{proposition}

\begin{proof}
    Let compact set $C \subset X$. Then $f(C) \subset Y$ is compact. Since compact set in Hausdorff is closed. $f\inv: Y \to X$ is continuous map. Then it's homeomorphic.
\end{proof}

\begin{proposition}
    Let $X$ be normal space. For $C \subset X$, give $\sim$ on $X$ by
    \[
        \begin{cases}
            x \sim y & \text{if } x, y \in C \\
            x \sim x & \text{otherwise}   
        \end{cases}
    \]
    is Hausdorff.
\end{proposition}

\begin{proof}
    Since $X$ is normal space, for $x \not\in C$, we have a disjoint closed set $\set{x} \in O_x \subset X$ and $C \subset O_C$. ($\because$ one point is closed in Hausdorff) Then, $[O_x], [O_C]$ in $X/{\sim}$.
\end{proof}

For instance, $D^2$ with given $\sim$ on $\partial{D^2}$. $D^2/{\sim}\cong S^2$.

\begin{proposition}
If $f$ is conti onto with $x\sim x' \implies f(x)=f(x')$ then
\[
\begin{tikzcd}
    X \arrow[r, "f"] \arrow[d, "q"'] & Y \\
    X/{\sim} \arrow[ur, "\bar f"']
\end{tikzcd}
\], $\bar f$ is conti onto.
\end{proposition}

\begin{definition}
    A topology space $X$ is called 'locally cpt' if $\forall x \in X$ there exists an open subset $O_x$ such that $x \in O \subset \overline{O_x}$ is compact.
\end{definition}

$\R$ is locally cpt.

\begin{theorem}
    Let $X$ be Hausdorff, locally compact, but not compact. Then there's a Hausdorff compact space $\widehat{X}$ (is called "one-point compactification of $X$") and a map $\iota: X\to \widehat{X}$ is 1-1 continuous such that
    \begin{enumerate}[label=\roman*)]
        \item $\widehat{X}\setminus \overline{\iota(X)}$ = single pt. (i.e. $\widehat{X}=\iota(X) \sqcup \set{\omega}$)
        \item $\iota$ sends open subsets to open.
    \end{enumerate}
\end{theorem}

\begin{proof}
    \begin{definition}
        $\widehat{X}=\iota(X) \sqcup \set{\omega}$ and given topology by
        \begin{enumerate}[label=\roman*)]
            \item open set $O \subset \widehat{X}\setminus\set{\omega}$ iff $O \subset X$.
            \item if $\omega \in O$, then $O=(X \setminus K) \sqcup \set{\omega}$, for compact set $K \subset X$.
        \end{enumerate}
    \end{definition}

    Then $\widehat{X}$ is compact (consider open cover s.t. $\omega \in O_\alpha$ and otherwise) and Hausdorff (set open set and using locally compact to build disjoint open set).
\end{proof}

Then, $\widehat{\R^2} = S^2 = D^2/{S^1}$. This can be shown by stereographic projection.

    \[
        \dim(\ker(ST)) \leq \dim(\ker(S)) + \dim(\ker(T))
    \]
\end{proposition}

\begin{proof}
    Consider $T|_{\ker(ST)}: \ker(ST) \to \ker(S)$. Then, $\ker(T|_{\ker(ST)})=\ker(T)$. By rank-nullity theorem, $\dim(\ker(ST))=\dim(\ker(T))+\dim(\im(T|_{\ker(ST)}))$. With $\im(\ker(T|_{\ker(ST)}) \subseteq \ker(S)$, we have 실베스터 부등식.
\end{proof}

\begin{definition}
    Let a sequence of continuous function $\set{f_n : X \to \R}$. We called it "uniformly converges" if $\forall \epsilon > 0 \; \exists N > 0$ such that
    \[
        n,m>N \implies \abs{f_n(x)-f_m(x)} < \eps, \forall x \in X
    \]
\end{definition}

\begin{proposition}
    If $f_n$ is uniformly converges, 
    \begin{enumerate}
        \item $\exists f: X \to \R$ s.t. $forall x \in X, f(x):=\lim f_n(x)$ 
        \item $f$ is continuous
    \end{enumerate}
\end{proposition}

We can derive it easily, from property of complete metric space. 

\begin{theorem}
    Let normal space $X$ and a closed subset $A \subset X$, continuous $f : A \to [0,1]$. Then $exists$ continuous function $F: X \to [0,1]$ s.t. $F|_A = f$.
    \label{theorem:tietz_extension}
\end{theorem}

\begin{proof}
    Let $C_1 := f\inv([0,1/3])$, $C_2 := f\inv([2/3,1])$. Then by Uryshon's Lemma, there's a continuous function $g_1: X \to [1/3,2/3]$ s.t. $g_1|_{C_1} = 1/3$ and $2/3$ for $C_2$. Then $\abs{f(x)-g(x)} \leq 1/3, \forall x \in A$. By continuing this for $h_1(x)=f(x)-g_1(x)$. We have a continuous function
    \[h_n : X \to [-\frac{2^{n-2}}{3^n}, \frac{2^{n-2}}{3^n}], \forall x \in A\]

    Then well-defined $F(x) := \sum^\infty_{n=1} g_n(x)$ satisfies the condition. ($\because$ for each $n<m$ $\abs{F_n(x)-F_m(x)} \leq \abs{q_{n+1}(x)}+\cdots+\abs{q_m(x)}\leq \epsilon$, it uniformly converges!)
\end{proof}

\begin{definition}
    Let a topology space $X$ and an open set $O \subset X$.
    \begin{enumerate}
        \item $S$ is not dense in $O$ if $\bar{S} \cap O \subsetneq O$
        \item $S$ is called 'no where dense' if $S$ is not dense in any open subset $O$ in $X$.
        \item A topology space $X$ is called $\textbf{the first category}$ if $X$ is a at most countable union of no where dense subsets.
        \item A topology space $X$ is called $\textbf{the second category}$ if $X$ is not the first category.
    \end{enumerate}
\end{definition}

For example, $\set{\text{pt}}$ is nowhere dense and $S=\Q$ is first category. ($\because \bigcup \set{q}$) But $\R$ is the second category. (not trivial)

\begin{theorem}
    Let a complete metric space $X$. Then, any open subset $O$ in $X$ is of the second category.
    \label{theorem:Baire_category}
\end{theorem}

For example, $\R^2 \neq \bigcup_n L_n$ (not union of countable line)

\begin{corollary}
    Any subset $A$ in $X$ containing an open subset $O$ is of second category.
\end{corollary}

\begin{theorem}
    $D: \F^n \times \cdots \times \F^n \to \F$ s.t.
    \begin{enumerate}
        \item 다중선형성: 각 열벡터에 대해서 선형.
        \item 교대성: 인접한 두 열벡터가 같으면 0.
        \item 정규화: $D(e_1, \cdots, e_n)=1$.
    \end{enumerate}
    의 함수는 유일하게 존재한다.
\end{theorem}

\begin{proof}
\[
D(A)=\sum_{i_1=1}^n \cdots \sum_{i_n=1}^n A_{i_1,1}\cdots A_{i_n,n} D(e_{i_1},\cdots,e_{i_n})
\]
교대성에 의해 인덱스 중에 중복되는 것이 하나라도 있으면 사라지기에, 중복이 없는 경우 즉 치환을 이루는 경우만 남는다.

\[
D(A)=\sum_{\sigma \in S_n} A_{\sigma(1),1}\cdots A_{\sigma(n),n} D(e_{\sigma(1)},\cdots,e_{\sigma(n)})
\]

이때 반대칭성을 적용하여 치환 $\sigma$를 항등치환으로 되돌리려면 부호 함수 $\sgn(\sigma)$가 곱해져야 한다.

\[
D(A)=\sum_{\sigma \in S_n} \sgn(\sigma)A_{\sigma(1),1}\cdots A_{\sigma(n),n}
\]

따라서 $D(A)$의 값은 행렬 $A$의 성분들에 의해 유일하게 결정된다.

\end{proof}

\begin{proposition}
    $\det(AB)=\det(A)\det(B)$
\end{proposition}

\begin{proof}
    Assume $B$ is not invertible. And define $f: M_{n \times n}(\F) \to \F$, with $f(A)=\det(AB)/\det(B)$. Then $f(I)=\det(B)/\det(B)=1$, and by uniqueness of determinant, $f(A)=\det(A) \implies \det(A)\det(B)=\det(AB)$.
\end{proof}

\begin{theorem}
    Let a complete metric space $X$. Then, any open subset $O$ in $X$ is of the second category.
\end{theorem}

\begin{proof}
    Suppoe $O=\bigcup_n S_n$ where $S_n$ is no where dense subset.

    Choose an open ball $O_1 = B_\rho(c_1) \subset O$ and an open ball $O_2 = B_{\frac{\rho}{2}}(c_2)$ with $\overline{O_2} \subset O_1 \setminus (\overline{S_1} \cap O_1)$. Continue this process. Then we have
    \[
        O_n = B_{\frac{\rho}{2^{n-1}}}(c_n)
    \] with $\overline{O_n} \subset O_{n-1}\setminus (\overline{S_{n-1}}\cap O_{n-1})$.

    \begin{enumerate}[label=\roman*.]
        \item $O_1 \supset O_2 \supset \cdots$
        \item Any two points in $O_n$ is closer than $\frac{\rho}{2^{n-1}}$.
        \item Any point in $O_n$ is outside of $S_1 \cup \cdots \cup S_{n-1}$.
    \end{enumerate}

    Then $\set{c_n}$ is Cauchy sequence, so in complete metric space, $c_n \to c \in X$.
    \begin{enumerate}[label=\roman*.]
        \item $c \in O$, but $c \not\in \bigcup_n S_n$
        \item $c \not\in O \implies c \not\in \overline{O_m}$ for all $m$, but $c_{m+1}, c_{m+2}, \cdots \in O_m$ then $c \in \overline{O_m}$
    \end{enumerate}

    By contradiction, we finish our proof.
\end{proof}

\begin{definition}

    \begin{enumerate}
        \item $X$ is Hausdorff.
        \item $\forall x \in X$, there's some open subset $U$ containing $x$ s.t. $U \simeq \R^n$ for some n.
    \end{enumerate}
\end{definition}

If $X$ is connected, then we have invariance of domain.

\begin{enumerate}
    \item $\R^0$ is single point. Then discrete topology space is 0-manifold.
    \item trivially $\R^n$ is n-manifold.
    \item and any open subset $U$ in $\R^n$ is n-dim manifold.

    \item an open subset of manifold is a manifold. ($O \cap U$ is open in $\R^n$ then $B_\delta(0)$ in it)

\begin{theorem}
    Let n-dim manifold M.
    $M$ is connected iff $X$ is path-connected.
\end{theorem}

\begin{proof}
    Pick $x_0 \in M$ and let subset $N$ s.t. $\forall y \in N$, there's a continuous map 
    \[
        f: I \to M
    \] with $f(0)=x_0 \wedge f(1)=y$. If we show this is closed and open, it becomes $M$ otherwise it is disconnected.

    $M$ satisfies first axiom of countability because it is locally $\R^n$. (to apply thm 5.1) Suppose $\set{x_k}$ is a sequence of point in $N$, $\lim x_k = z$. Choose an open neighbor of $z$ which homeomorphic to $\R^n$. For large $k$, $x_k \in O$. As $x_k \in N$, we have $f: I \to M$. And since $\R^n$ is locally path-connected, choose continuous map $g: I \to M$. Then define $h$ by
    \[
        h(t)=\begin{cases}
            f(2t) & 0 \leq t \leq 1/2 \\
            g(2t-1) & 1/2 < t \leq 1 \\
        \end{cases}
    \] So $z \in N$. $N$ is closed.

    Likewise, $x \in N$ there's open set $O$ homeomorphic to $\R^n$. Then every element is in $N$. $N$ is open.
\end{proof}

\begin{theorem}
    \label{theorem:taylor}
    Let $f:[a,b] \to \R$. Suppose $f^{n-1}$ is conti on $[a,b]$ and differentiable in $(a,b)$. Let $\alpha, \beta \in [a,b]$, $\alpha < \beta$ and
    \[
    P(t)=\sum_{k=0}^{n-1} \frac{f^{k}(\alpha)}{k!}(t-\alpha)^k
    \]. Then there's $c \in (\alpha, \beta)$ s.t. \[f(\beta)-P(\beta)=\frac{f^n(c)}{n!}(\beta-\alpha)^n\]
\end{theorem}

\begin{proof}
    $f(\beta)=P(\beta)+M(\beta-\alpha)^n$일 때, \[g(t)=f(t)-P(t)-M(t-\alpha)^n\]라고 두자. 이때 $i=0,\cdots,n-1$에 대해서 $f^{(i)}(\alpha)-P^{(i)}(\alpha)=0$이기에 $g^{(i)}(\alpha)=0$이다. $g(\beta)=0$이고 평균값 정리를 쓰면 $g'(c_1)=0$인 $c_1$가 있다. 이를 반복하면 $g^{(n)}(c_n)=0$인 $c_n$이 있다. $f^{(n)}(c_n)-n!M=0$이고 $M=\frac{f^{(n)}(c_n)}{n!}$이다. 
\end{proof}

\subsubsection{Riemann and Stieltjes Integral}

일단 partition을 정의하자. $P=\{x_0, x_1, \cdots, x_n\}$ s.t. $a=x_0 < x_1 < \cdots < x_n=b$ 
set $\Delta x_n := x_n - x_{n-1}$ 

let $f$ be a bounded ftn $[a,b] \to \mathbb{R}$

\[
M_i := \sup_{[x_{i-1}, x_i]} f
\]
similarly for $m_i$

\[
U(P, f)=\sum M_i \Delta x_i, \overline{\int^b_a}f(x)\dd x := \inf_P U(P,f)
\]
similarly for $L(P, f)$.

\begin{definition}
    We say $f$ is Riemann integrable if $\overline{\int^b_a} f(x)\dd x = \underline{\int^b_a} f(x)\dd x$.
\end{definition}

ex) $f=x_{\mathbb{Q}}$ -> $U(P, f)=1 \neq L(P, f)=0$\\
check) $f=x_{contour}$

\begin{definition}
    Let $\alpha$ be a monotonically increasing ftn on $[a, b]$ for a partition $P$.\\$\Delta \alpha_i = \alpha(x_i)-\alpha(x_{i-1}) \geq 0$
    \[M_i=\sup_{x \in [\alpha(x_{i-1}), \alpha(x_{i})]} f(x)\]
    \[U(P, f, \alpha)=\sum M_i \Delta \alpha_i\]
    \[\overline{\int^b_a}f(\alpha)\dd\alpha := \inf_P U(P, f, \alpha)\]
\end{definition}

alpha에 대한 적분으로 바꾸자.

\begin{definition}
    We say $f \in R(\alpha)$ (is Riemann integrable with respect to $\alpha$) if $\overline{\int^b_a}f(\alpha)\text{d}\alpha=\underline{\int^b_a}f(\alpha)d\alpha$
\end{definition}

\begin{definition}
    Let $P, P'$ be partitions of $[a, b]$, We say $P'$ is refinement of P if $P \subset P'$
\end{definition}

\begin{theorem}
    if $P'$ is refinement of P, then $U(P', f, a) \leq U(P, f, \alpha)$
\end{theorem}

\begin{theorem}
    $f \in R(\alpha)$ on $[a, b]$ iff for every $\epsilon >0$ . there's parition P s.t. $U(P, f, \alpha)-L(P, f, \alpha) < \epsilon$
\end{theorem}
\begin{proof}
    (only if) Let $\epsilon>0$ 그냥 생각해보면 됨. inf/sup 정의로 epsilon만들기 (Partition $P_1 \cup P_2$를 생각)
    (if) 계속 작아지니까 결국 같아짐. (완비성, $\Q$로 가는 함수라면 다르다!)
\end{proof}

\end{theorem}
