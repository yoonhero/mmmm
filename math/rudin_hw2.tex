\subsection*{5.}

\begin{lemma}
    Let $b_n=\sup_{m \geq n} a_m$, then
    \[
    \limsup_{\ntoinf} a_n = \lim_{\ntoinf} b_n
    \]
    \label{lemma:limsup}
\end{lemma}

\begin{proof}
    The $b_n$ is monotically decreasing sequence. ($\because$ $\{a_k\}_{k \geq n+1} \subset \{a_k\}_{k \geq n}$) In extended real, $L:=\lim b_n=\inf b_n$. 

    Set a subsequence $\{a_{n_j}\}$ converges to $c$ s.t. $n_j > N$, $a_{n_j} \leq b_N$ since $a_{n_j} \in \{a_k\}_{k \geq N}$. $c \leq b_N$, then $c \leq \inf b_n = L$ Then $\limsup a_n \leq L$.\\
    For sufficiently large $k$, $b_n < L + 1/k$ for $n \geq N_k$. Since $b_n = \sup_{m \geq N_k} a_m$, $\exists n_k > N_k > k$,
    
    \[
        a_{n_k}>b_{N_k}-1/k \implies a_{n_k}>L-1/k
    \]

    \[
        L+1/k \geq b_{N_k} \geq a_{n_k}
    \]

    Then, 
    \[
        0 \leq d(a_{n_k}, L) < 1/k
    \]

    Then $L \leq \limsup a_n$. Therefore $\limsup a_n=\lim b_n$
\end{proof}

For $n$, $\sup_{m \geq n}(a_m+b_m) \leq \sup_{m \geq n}(a_m) + \sup_{m \geq n}(b_m)$. ($\because \, a_m \leq \sup_{m \geq n}(a_m)$ similarly $b_m$ then its sum is bounded) With \cref{lemma:limsup}, we have $\limsup(a_n+b_n)\leq\lim(\sup(a_m) + \sup(b_m))=\limsup(a_n)+\limsup(b_n)$ ($\because$ it is not form of $\infty - \infty$)

\subsubsection*{e}

If $m < n$, then 

\[
\frac{m+1}{n-m}(\sigma_n-\sigma_m)+\frac{1}{n-m} \sum_{i=m+1}^n (s_n - s_i) = s_n+\frac{m+1}{n-m}\sigma_n - \frac{s_0+s_1+...+s_{n}}{n-m}=s_n-\sigma_n
\]

From $\abs{na_n} \leq M$, $\abs{s_n-s_{n-1}}<M/n$ for $n>0$. For these $i$,

\[
|s_n-s_i|\leq \abs{s_n-s_{n-1}}+...+\abs{s_{i+1}-s_i} \leq \frac{(n-i)M}{i+1} \leq \frac{(n-m-1)M}{m+2}
\]

Fix $\eps > 0$ and associate with each $n$ the integer $m$ that satisfies

\[
m \leq \frac{n-\eps}{1+\eps} < m+1
\]

Then $(m+1)/(n-m) \leq 1/\eps$, $\abs{s_n-s_i} < M\eps$ and $\abs{\sigma_n-\sigma_m}\leq \eps^2$. For sufficiently large $n$, 

\[
\abs{s_n-\sigma_n} \leq \frac{1}{\eps}\eps^2 + \frac{n-m}{n-m} \frac{(n-m-1)M}{m+2} \leq \eps + M\eps
\]

With this,

\[
\abs{s_n-\sigma} \leq \abs{s_n-\sigma_n} + \abs{\sigma_n-\sigma} \leq (M+1)\eps + \eps.
\]

Sicne $\eps$ was arbitrary, $lim s_n = \sigma$.
